\chapter{Algebra Omologica}

Sia \(\mathcal{A}\) una categoria preadditiva. Indichiamo un complesso in
\(\mathcal{A}\) con 
\[
  X^\bullet := {(\dots \to X^{i-1} \overset{d_{X^\bullet}^{i-1} }{\to} X^{i}
  \overset{d_{X^\bullet}^{i}}{\to} X^{i+1} \to \dots)}
\]
dove \(d^{i} \circ d^{i-1} = 0, \;\forall i \in \mathbb{Z}\).
Spesso quasi tutti gli \(X^{i}\) sono nulli e la catena è dunque limitata.

\section{Complessi e coomologia}

\begin{definition}{Categoria dei Complessi}
    La \textbf{categoria dei complessi} \(C{(\mathcal{A})}\) ha come oggetti i
    complessi di \(\mathcal{A}\) e morfismi
    \[
      C{(\mathcal{A})}{(X^\bullet, Y^\bullet)} := \{f^\bullet = {(f^{i} \in
      \mathcal{A}{(X^{i}, Y^{i})})}_{i \in \mathbb{Z}} : d_Y^{i} \circ f^{i} =
  f^{i+1} \circ d_X^{i} \quad \forall i \in \mathbb{Z}\} 
    \]
    Ossia, il seguente diagramma commuta:

    \[\begin{tikzcd}
        \dots & Y^{i-1} & Y^{i} & Y^{i+1} & \dots\\
        \dots & X^{i-1} & X^{i} & X^{i+1} & \dots
        \arrow[from=1-1, to=1-2]
        \arrow["{d_Y^{i-1}}", from=1-2, to=1-3]
        \arrow["{d_Y^i}", from=1-3, to=1-4]
        \arrow[from=1-4, to=1-5]
        \arrow[from=2-1, to=2-2]
        \arrow["{f^{i-1}}", from=2-2, to=1-2]
        \arrow["{d_X^{i-1}}"', from=2-2, to=2-3]
        \arrow["{f^i}", from=2-3, to=1-3]
        \arrow["{d_X^i}"', from=2-3, to=2-4]
        \arrow["{f^{i+1}}", from=2-4, to=1-4]
        \arrow[from=2-4, to=2-5]
    \end{tikzcd}\]

    e la composizione \(f^\bullet \circ g^\bullet = {(f^{i} \circ g^{i})}_{i \in
    \mathbb{Z}}\) è definita componente per componente 

\end{definition}

\begin{remark}{}
    \(C{(\mathcal{A})}\) è sottocategoria piena di \(S{(\mathcal{A})}\) la
    categoria delle successioni senza la condizione \(d^{i} \circ d^{i-1} = 0\)
    e i morfismi definiti nello stesso modo; \(S{(\mathcal{A})} \cong
    \mathtt{Fun}{({(\mathbb{Z}, \le )}, \mathcal{A})}\). Dunque poiché
    \(\mathcal{A}\) è preadditiva, anche \(S{(\mathcal{A})}\) lo è e quindi
    anche \(C{(\mathcal{A})}\). Inoltre se \(\mathcal{A}\) è additiva o
    (pre)abeliana allora anche \(C{(\mathcal{A})}\) lo è. Infatti
    \(\mathcal{C}{(\mathcal{A})}\) è chiusa per (co)prodotti finiti, nuclei e
    conuclei, che esistono in \(S{(\mathcal{A})}\).
\end{remark}

Se $\mathcal{A}$ è una categoria abeliana, per ogni intero $i \in \mathbb{Z}$
possiamo definire un funtore ``nucleo'', denotato con $K^i: C(\mathcal{A}) \to
\mathcal{A}$ come $K^i(X^\bullet) = \operatorname{Ker}(d_X^i)$.
L'azione sui morfismi è indotta naturalmente: dato un morfismo di complessi
$f^\bullet: X^\bullet \to Y^\bullet$, esiste un unico morfismo $K^i(f)$ che
rende commutativo il seguente diagramma:


\[\begin{tikzcd}
    \mathrm{Ker}d_Y^{i} & Y^{i} & Y^{i+1}\\
    \mathrm{Ker}d_X^{i} & X^{i} & X^{i+1}
	\arrow["{\mathrm{ker}d_Y^i}", tail, from=1-1, to=1-2]
	\arrow["{d_Y^i}", from=1-2, to=1-3]
	\arrow["{\exists!K^i(f)}", from=2-1, to=1-1]
	\arrow["{\mathrm{ker}d_X^i}"', tail, from=2-1, to=2-2]
	\arrow["{f^i}"', from=2-2, to=1-2]
	\arrow["{d_X^i}"', from=2-2, to=2-3]
	\arrow["{f^{i+1}}"', from=2-3, to=1-3]
\end{tikzcd}\]

per le proprietà del \(\mathrm{Ker}\).

Analogamente, possiamo definire un funtore ``immagine''
\begin{align*}
    I^{i}: C{(\mathcal{A})} &\longrightarrow \mathcal{A} \\
     X^\bullet &\longmapsto I^{i}(X^\bullet) = \mathrm{Im}{(d_X^{i-1})}
\end{align*}

Dove sui morfismi è definito grazie alla scomposizione in mono + epi di
\(d^{i-1}\)\footnote{non sono sicuro di questa cosa, mi è stata rigurgitata da
    Gemini e mi sembra abbia senso ma non sono sicuro, inoltre potrebbero
esserci metodi più facili e chiari. L'idea di base è che sia la categorizzazione
della restrizione di \(f\) sull'immagine di \(d\), ossia \(I^{i}{(f)} =
f^{i+1}|_{\mathrm{Im}(d_X^{i})} : \mathrm{Im}{(d_X^{i})} \to
\mathrm{Im}{(d_Y^{i})} \) che è ben definita perché \(f^\bullet\) è morfismo di
complessi e quindi \(f^{i+1}{(d_X^{i}{(x)})} = d_Y^{i}{(f^{i}{(x)})}\) } 

$$ \begin{tikzcd}[column sep=large, row sep=large]
    X^{i-1} \arrow[r, "\pi_X", two heads] \arrow[d, "f^{i-1}"'] &
    I^i(X^\bullet) \arrow[r, "\iota_X", tail] \arrow[d, "\exists! I^i(f)", dashed] & 
    X^i \arrow[d, "f^i"] \\
    Y^{i-1} \arrow[r, "\pi_Y"', two heads] & I^i(Y^\bullet) \arrow[r,
    "\iota_Y"', tail] & Y^i
\end{tikzcd} $$

Poiché \(d^{i-1} \circ d^{i} = 0\), \(\mathrm{Im}{(d_X^{i-1})} <
\mathrm{Ker}{(d_X^{i})}\) come sottoggetti, dunque la seguente definizione ha
senso.
\begin{definition}{Funtore di coomologia}
    Esiste una successione esatta in \(\mathcal{A}\) \[ 0 \to I^{i}{(X^\bullet)}
    \to K^{i}{(X^\bullet)} \to H^{i}{(X^\bullet)} \to 0\]
    Dove \(H^{i}\) così definito è
    \begin{align*}
        H^{i}: C{(\mathcal{A})} &\longrightarrow \mathcal{A} \\
        X^\bullet &\longmapsto H^{i}(X^\bullet) =
        \frac{K^{i}{(X^\bullet)}}{I^{i}{(X^\bullet)}}
    \end{align*}
    l'\(i\)-esimo \emph{funtore di coomologia} di \(\mathcal{A}\).
\end{definition}

\begin{proposition}{}
    Tutti i funtori \(H^{i}\), \(K^{i}\) e \(I^{i}\) sono naturalmente
    additivi.
\end{proposition}

Sia \(\mathcal{A}\) una categoria preadditiva, e \(C{(\mathcal{A})}\) la
categoria dei complessi. Siano \(X^{\cdot }\) e \(Y^{\cdot } \in
C{(\mathcal{A})}\). Allora dati \(\kappa^{i} \in \mathcal{A}{(X^{i}, Y^{i-1})}\)
per ogni \(i \in \mathbb{Z}\), si definisca \(f^\bullet\) come
\begin{equation}{}\label{eq:omotopia-complessi}
    f^{i} := d_Y^{i-1} \circ \kappa^{i} + \kappa^{i+1} \circ d_X^{i} 
\end{equation}

Allora segue che \(f^\bullet \in C{(\mathcal{A})}{(X^\bullet, Y^\bullet)}\),
infatti
\begin{align*}{}
   d_Y^{i} \circ f^{i} &= d_Y^{i} \circ \kappa^{i+1} \circ
   d_X^{i} = f^{i+1} \circ d_X^{i} = \\&={(d_Y^{i} \circ \kappa^{i+1} + \kappa^{i+2}
   \circ d_X^{i+1})} \circ d_X^{i} = f^{i+1} \circ d_X^{i}
\end{align*}


\begin{definition}{Omotopia}
    \(f^\bullet \in C{(\mathcal{A})}{(X^{\bullet }, Y^{\bullet})}\) è
    \emph{omotopo a 0} se \(\exists k^{i} \in \mathcal{A}{(X^{i}, Y^{i-1})}\)
    tale che vale~\eqref{eq:omotopia-complessi}.

    \(f^{\bullet}\) e \(g^\bullet \in C{(\mathcal{A})}{(X^\bullet, Y^\bullet)}\)
    sono \emph{omotopi} (denotato con \(f^\bullet \sim g^\bullet\)) se
    \(f^\bullet - g^\bullet \) è omotopo a 0.
\end{definition}

\begin{remark}{} 
    \(\mathtt{Htp}{(X^\bullet, Y^\bullet)} := \{f^\bullet \in C{(\mathcal{A})}
    {(X^\bullet, Y^\bullet)} | f^\bullet \sim 0\} \). Allora \(\mathtt{Htp}
    \subseteq C{(\mathcal{A})} \) è ideale, ossia \(\mathtt{Htp}{(X^\bullet,
    Y^\bullet)} < C{(\mathcal{A})}{(X^\bullet, Y^\bullet)}\) e \(g^\bullet \circ
    f^\bullet \in \mathtt{Htp}{(X^\bullet, Z^\bullet)}\) se \(f^\bullet \in
    \mathtt{Htp}{(X^\bullet, Y^\bullet)}\) o \(g^\bullet \in
    \mathtt{Htp}{(Y^\bullet, Z^\bullet)}\).

    Questo vale perché se \(f^\bullet \sim 0\) con \(\kappa^{i}\), allora
    \(g^\bullet \circ f^\bullet \sim 0\) con \(\overline{\kappa}^{i} = g^{i-1}
    \circ \kappa^{i}\) 
\end{remark}

Come conseguenza si ottiene una categoria preadditiva \(K{(\mathcal{A})} :=
C{(\mathcal{A})} / \mathtt{Htp}\) (categoria \emph{omotopa} di \(\mathcal{A}\))
con funtore additivo \(Q = Q_{\mathcal{A}} : C{(\mathcal{A})} \to
K{(\mathcal{A})}\) tale che \(X^\bullet \mapsto X^\bullet\) e \(f^\bullet
\mapsto [f^\bullet]\) 



\begin{remark}{}
    Si dimostra che se \(\mathcal{A}\) è abeliana, allora 
    \[
      \mathcal{A} \text{ è semisemplice } \iff K{(\mathcal{A})}\text{ è abeliana
      } \iff K{(\mathcal{A})}\text{ è preabeliana }
    \]
\end{remark}

\begin{remark}{}
    Sia \(F: \mathcal{A} \to \mathcal{B}\) un funtore additivo tra categorie
    preadditive. Allora \(F\) induce funtori \(C{(F)} : C{(\mathcal{A})} \to C
    {(\mathcal{B})}\) e \(K{(F)} : K{(\mathcal{A})} \to K{(\mathcal{B})}\) tali
    che, per ogni \(X \in C{(\mathcal{A})}\), \(C{(F)} {(X^\bullet)} =
    K{(F)}{(X^\bullet)} = \) 
    \[
       = {\left( \dots \to F{(X^{i-1})} \overset{F{(d^{i-1})}}{\to} F{(X^{i})}
       \overset{F{(d^{i})}}{\to} F{(X^{i+1})} \right)} 
    \]
    e per funtorialità \(F{(d^{i})} \circ F{(d^{i-1})} = F{(d^{i} \circ
    F^{i-1})} = F{(0)} = 0\) perché è additivo.

    E esplicitamente \(\forall f^\bullet : X^\bullet \to Y^\bullet\),
    \(C{(F)}{(f^\bullet)}^{i} := F{(f^{i})}\) e \(K{(F)} {([f^\bullet])} :=
    [C{(F)}{(f^\bullet)}]\) (mostrare per esercizio che è sempre ben
    definito). Di solito si scrive direttamente \(F\) invece di \(C{(F)}\) o
    \(K{(F)}\) 
\end{remark}

\begin{lemma}{}
    Sia \(\mathcal{A}\) una categoria abeliana. Sia \(f^\bullet \in
    \mathtt{Htp}{(X^\bullet, Y^\bullet)}\). Allora \(H^{i}{(f^\bullet)} = 0\)
    per ogni \(i \in Z\) 
\end{lemma}
\begin{proof}{}
    \(f^{i} = d_{Y^\bullet} ^{i-1} \circ k^{i} + k^{i+1} \circ
    d_{X^\bullet}^{i}\). Osservando il diagramma a righe esatte
\[ \begin{tikzcd}
	0 & \mathrm{Im}{(d_{X^\bullet} ^{i-1})} & \mathrm{Ker}{(d_{X^\bullet} ^{i})}
      & H^{i}{(X^\bullet)} & 0 \\
	0 & \mathrm{Im}{(d_{Y^\bullet} ^{i-1})} & \mathrm{Ker}{(d_{Y^\bullet} ^{i})}
      & H^{i}{(Y^\bullet)} & 0
	\arrow[from=1-1, to=1-2]
	\arrow[from=1-2, to=1-3]
	\arrow["I^{i}{(f^\bullet)}"', from=1-2, to=2-2]
	\arrow[from=1-3, to=1-4]
	\arrow["K^{i}{(f^\bullet)}", from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow["{H^i(f^\bullet)}", dashed, from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow[from=2-2, to=2-3]
	\arrow[from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd} \]

Ma allora \(H^{i}{(f^\bullet)} = 0 \iff K^{i}{(f^\bullet)}\) fattorizza
attraverso \(I^{i}{(Y^\bullet)} \to K^{i}{(Y^\bullet)}\) se e solo se
\(\mathrm{Im}{(K^{i}{(f^\bullet)})} \subseteq \mathrm{Im}{(d_{Y^\bullet}
^{i-1})} \). Allora \(K^{i}{(f^\bullet)}\) è indotto da \(f^{i}\), quindi da
\(d_{Y^\bullet} ^{i-1} \circ K^{i}\) (perché \(d_{X^\bullet} ^{i}\) è 0 su
\(K^{i}{(f^\bullet)}\)).
\end{proof} 

\begin{corollary}{}
    Sia \(\mathcal{A}\) abeliana. Allora \(\forall i \in \mathbb{Z}\) il funtore
    \(H^{i} : C{(\mathcal{A})} \to \mathcal{A}\) induce un funtore additivo
    \(H^{i} : K{(\mathcal{A})} \to \mathcal{A}\) 
\end{corollary}

\section{Funtori derivati}

\begin{definition}{Risoluzione}
    Sia \(\mathcal{A}\) una categoria abeliana. Una \emph{risoluzione} a
    \emph{sinistra}/\emph{destra} di un oggetto \(X \in \mathcal{A}\) è una
    successione esatta 
    \[
     \dots \to X^{-2} \to X^{-1} \to X^{0} \to X \to 0 / 0 \to X \to X^{0} \to
     X_{1} \to X_{2} \to \dots
    \] 
    (di solito si scrive \(\dots\to X_2 \to X_{1} \to X_{0} \to X \to
    0\) nel caso di risoluzione sinistra).

    Inoltre una tale soluzione ha \emph{lunghezza} \(n\) se (ad esempio nel caso
    di risoluzione destra)
    \[
      X^{n} \neq 0 \text{ e } {(X^{i} = 0)} \quad \forall i > n
    \]
    Se tale \(n\) non esiste si dice che la lunghezza è \(\infty\) 
\end{definition}

Esistono diversi tipo di risoluzione a seconda del tipo di problemi che si sta
affrontando. Per i nostri scopi vedremo di studiare gli oggetti iniettivi e
suriettivi.

\begin{definition}{}
    Una risoluzione \emph{proiettiva}/\emph{iniettiva} di \(X \in \mathcal{A}\)
    è una risoluzione \emph{sinistra}/\emph{destra} \emph{con} \(X_{i}\)
    \emph{proiettivo} / \emph{con }\(X^{i}\) \emph{iniettivo} per ogni \(i \in
    \mathbb{N}\).
\end{definition}

\begin{proposition}{}
    Ogni \(X \in \mathcal{A}\) ha una risoluzione
    \emph{proiettiva}/\emph{iniettiva} se e solo se \(\mathcal{A}\) ha
    abbastanza \emph{proiettivi}/\emph{iniettivi}.
\end{proposition}
\begin{proof}\( \)
\begin{itemize}
    \item[\(\implies \)] ovvio.
    \item[\(\impliedby \)] \(\exists d^{0} : X^{0} \to X\) epimorfismo, con
        \(X^{0}\) proiettivo. Allora \(\exists \tilde{d}^{1} : X^{1} \to
        \mathrm{Ker}{(d^{0})}\) epi con \(X^{1}\) proiettivo. Allora \(d^{1} :=
        \mathrm{Ker}{(d^{0})} \circ \tilde{d}^{1} : X^{1} \to X^{0}\) e si può
        proseguire per induzione.
\end{itemize}
\end{proof}

\begin{proposition}{}\label{prop:non-so-niente}
    Sia \(\mathcal{A}\) una categoria abeliana e sia \(f : X \to Y\) un morfismo
    di \(\mathcal{A}\). E siano
    \[ \begin{tikzcd}
	0 & X & X^{0} & X^{1} & \dots \text{ risoluzione di \(X\) }\\
    0 & Y & I^{0} & I^{1} & \dots \text{ complesso con ogni \(I^{i}\) iniettivo}
	\arrow[from=1-1, to=1-2]
	\arrow["s", from=1-2, to=1-3]
	\arrow["f"', from=1-2, to=2-2]
	\arrow["d^{0}", from=1-3, to=1-4]
	\arrow["f^{0}", dashed, from=1-3, to=2-3]
	\arrow["d^{1}", from=1-4, to=1-5]
	\arrow["f^{1}", dashed, from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow["t",from=2-2, to=2-3]
	\arrow["\ell^{0}", from=2-3, to=2-4]
	\arrow["\ell^{1}", from=2-4, to=2-5]
\end{tikzcd} \]
    Si denoti inoltre \(X^\bullet := {\left( 0 \to X^{0} \overset{d^{0}}{\to}
    X^{1} \to \dots \right)} \) e \(I^\bullet := {\left( 0 \to I^{0}
\overset{\ell^{0}}{\to} I^{1} \to \dots \right)} \).
    Allora
\begin{enumerate}[label = \arabic*)]
    \item \(\exists \, f : X^\bullet \to I^\bullet\) in \(C{(\mathcal{A})}\) che
        estende \(f\), cioè \(f^0 \circ s = t \circ f\) 
    \item \([f^\bullet]\) è unico in \(K{(\mathcal{A})}\) 
\end{enumerate}
\end{proposition}
\begin{proof}{}
    Siano \(\tilde{X}^\bullet\) e \(\tilde{I}^\bullet\) i complessi delle righe
    del diagramma. In particolare \(\tilde{X}^{-1} = X\), \(\tilde{X}^{i} =
    X^{i}\) per ogni \(i \in \mathbb{N}\) e \(\tilde{X}^{i} = 0\) per ogni \(i <
    -1\). Similmente per \(\tilde{I}^{\bullet}\). Inoltre si denoti anche \(s =
    d^{-1}\) e \(t = \ell^{-1}\).

\begin{enumerate}[label = \arabic*)]
    \item Cerco \(\tilde{f}^\bullet : \tilde{X}^\bullet \to \tilde{I}^\bullet\)
        in \(C{(\mathcal{A})}\) tale che \(\tilde{f}^{-1} = f\). Dato \(n \in
        \mathbb{N}\) e supponendo di avere già definito \(\tilde{f}^{i} \) per
    ogni \(i < n\) tale che \(\tilde{f}^{i} \circ d^{i-1} = \ell^{i-1} \circ
    \tilde{f}^{i-1}\) per ogni \(i < n\).

    Cerco ora \(f^{n} = \tilde{f}^{n} : X^{n} \to I^{n}\) tale che \(f^{n} \circ
    d^{n-1} = \ell^{n-1} \circ \tilde{f}^{n-1}\).
    \[ \begin{tikzcd}
	\tilde{X}^{n-2} & \tilde{X}^{n-1} & \tilde{X}^{n} = X^{n} \\
	& \tilde{I}^{n-1} & I^{n}
	\arrow["{d^{n-2}}", from=1-1, to=1-2]
	\arrow["{d^{n-1}}", from=1-2, to=1-3]
	\arrow["{\tilde{f}^{n-1}}"', from=1-2, to=2-2]
	\arrow[from=1-2, to=2-3]
	\arrow["{f^n}", dashed, from=1-3, to=2-3]
	\arrow["{\ell^{n-1}}"', from=2-2, to=2-3]
\end{tikzcd} \]
    è esatto, dunque \(I^{n}\) è iniettivo. Allora
    \[
      \mathcal{A}{(X^{n}, I^{n})} \overset{d^{n-1 *}}{\to}
      \mathcal{A}{(\tilde{X}^{n-1}, I^{n})} \overset{d^{n-2*}}{\to}
      \mathcal{A}{(\tilde{X}^{n-2}, I^{n})}
    \] è esatta, e quindi esiste \(f^{n} \in \mathcal{A}{(X^{n}, I^{n})}\) tale
    che 
    \begin{align*}
      &d^{n-1*}{(f^{n})} = f^{n} \circ d^{n-1} = \ell_{n-1}  \circ
      \tilde{f}^{n-1} \iff \\ \iff &0 = d^{n-2*}{\left( \ell^{n-1} \circ \tilde{f}^{n-1}
      \right)} = \ell^{n-1} \circ \tilde{f}^{n-1} \circ d^{n-2} =
      \underbrace{\ell^{n-1} \circ \ell^{n-2}}_{0} \circ \tilde{f}^{n-2}
    \end{align*}

    \item Dati \(f^\bullet, g^\bullet : X^\bullet \to I\) che estendono \(f\) ,
        devo dimostrare che \(f^\bullet \sim g^\bullet\). Basta allora
        dimostarre che dato \(h^\bullet : X^\bullet \to I^\bullet\) che estede
        \(0 : X \to Y\), allora \(h^\bullet \sim 0\).
        A tale scopo, sia \(\tilde{h}^\bullet : \tilde{X}^\bullet \to
        \tilde{I}^\bullet\) dato da \(\tilde{h}^{i} := h^{i}\), \(\forall i \in
        \mathbb{N}\) e \(\tilde{h}^{i} := 0\), \(\forall i < 0\). Allora
        \(\tilde{h}^\bullet\) è un morfismo in \(C{(\mathcal{A})}\). Voglio
        dunque dimostrare che \(\exists k^{i} : \tilde{X}^{i} \to
        \tilde{I}^{i-1}\) tale che 
        \begin{equation}\label{eq:helpext1}
          \tilde{h}^{i} = \ell^{i-1} \circ k^{i} + k^{i+1} \circ d^{i} \quad
          \forall i \in \mathbb{Z}\text{ , con \(k^{0} = 0\)}
        \end{equation}
        (infatti da questo seguirebbe che \(\tilde{h}^\bullet \sim 0 \) e
        \(h^\bullet \sim 0\)).

        Dato \(n \in \mathbb{N}\) e assumendo di avere già definito i \(k^{i}\)
        per ogni \(i \le n\) tale che~\eqref{eq:helpext1} valga per ogni \(i <
        n\). Cerco allora \(k^{n+1} : X^{n+1} \to I^{n}\) tale che
        valga~\eqref{eq:helpext1} per \(i = n\). Allora il seguente diagramma
        con prima riga esatta commuta:
    \[ \begin{tikzcd}
	\tilde{X}^{n-1} & {X}^{n} & {X}^{n+1} \\
	& I^{n}
	\arrow["{d^{n-1}}", from=1-1, to=1-2]
	\arrow["{d^{n}}", from=1-2, to=1-3]
	\arrow["{h^{n} - \ell^{n-1} \circ k^{n}}"', from=1-2, to=2-2]
	\arrow["{k^{n+1}}", dashed, from=1-3, to=2-2]
\end{tikzcd} \]
    e come prima dunque \(\exists \, k^{n+1}\) tale che
    valga~\eqref{eq:helpext1} per \(i = n\) se e solo se 
    \begin{align*}
        0 &= \ell^{n-1} \circ k^{n} \circ d^{n-1} = \ell^{n-1} \circ {\left(
        \tilde{h}^{n-1} - \ell^{n-2} \circ k^{n-1} \right)} = \\ &= \tilde{h}^{n} \circ
      d^{n-1} - \ell^{n-1} \circ \tilde{h}^{n-1} = 0
    \end{align*}
\end{enumerate}
\end{proof}

\begin{corollary}{}\label{cor:injres_unique}
    Una risoluzione iniettiva di \(X \in \mathcal{A}\) categoria abeliana (se
    esiste) è unica a meno di isomorfismo in \(K{(\mathcal{A})}\).
\end{corollary}
\begin{proof}{}
    Siano \(0 \to X \to I^\bullet\) e \(0 \to X \to J^\bullet\) risoluzioni
    iniettive di \(X\). Allora applico la proposizione~\ref{prop:non-so-niente}
    con \(f = 1_X\), dunque \(\exists \, f^\bullet : I^\bullet \to J^\bullet\)
    che estende \(1_X\). Analogamente \(\exists \, g^\bullet : J^\bullet \to
    I^\bullet\) che estende \(1_X\). Segue che anche \(g^\bullet \circ f^\bullet
    : I^\bullet \to I^\bullet\) estende \(1_X\), ma anche \(1_{i^\bullet} \)
    estende \(1_X\), da cui \(g^\bullet \circ f^\bullet \sim 1_{I^\bullet} \) e
    analogamente \(f^\bullet \circ g^\bullet \sim 1_{J^\bullet} \) da cui
    \([f^\bullet]\) è isomorfismo.
\end{proof}

\begin{corollary}{}
    Sia \(\mathcal{A}\) una categoria abeliana con abbastanza iniettivi. Allora
    c'è un funtore ``risoluzione iniettiva'' \(I_\mathcal{A} : \mathcal{A} \to
    K(\mathcal{A})\) tale che \(I_{\mathcal{A}} {(X)} := I^\bullet\) e \(0 \to X
    \to I^\bullet\) è una risoluzione iniettiva (AoC per poterla scegliere per
    ogni \(X \in \mathcal{A}\)).

    Inoltre \(\forall f : X \to Y\) in \(\mathcal{A}\), \(I_{\mathcal{A}} {(f)}
    := [f^\bullet]\) con \(f^\bullet\) che estende \(f\). Si ottiene dunque un
    funtore (esercizio) la cui classe di isomorfismo non dipende dalle scelte.
\end{corollary}

\begin{definition}{Funtore derivato (destro)}
    Sia \(F : \mathcal{A} \to \mathcal{B}\) un funtore additivo tra categorie
    abeliane. Sia \(\mathcal{A}\) con abbastanza iniettivi. Allora
    \(I_\mathcal{A}\) è ben definito e, per \(i \in \mathbb{N}\), la composizione 
    \[
      R^{i}F : \mathcal{A} \overset{I_\mathcal{A}}{\to} K{(\mathcal{A})}
      \overset{F}{\to} K{(\mathcal{B})} \overset{H^{i}}{\to} \mathcal{B}
    \]
    è l'\(i\)-esimo \textbf{funtore derivato destro} di \(F\) 
\end{definition}
\begin{remark}{}
    Poiché è composizione di funtori additivi, anche ogni \(R^{i}F\) è additivo.
    Inoltre per il corollario~\ref{cor:injres_unique} \(R^{i}F\) è ben definito
    (a meno di isomorfismo).
\end{remark}

Non ci si mette quasi mai a calcolare esplicitamente questi funtori.

    
\begin{example}[\(R^{i}F{(I)}\), con \(I\) iniettivo]
    Se \(I\) è iniettivo, allora una risoluzione iniettiva è \(0 \to I
    \overset{1_I}{\to} I \to 0\). Segue che 
    \[
      R^{i}F{(I)} = H^{i}{(F{(0 \to I \to 0)})} = H^{i}{(0 \to F{(I)} \to 0)} =
      \begin{cases}{}
          F{(I)} & i = 0\\
          0      & i \neq 0
      \end{cases}
    \]
\end{example}


\begin{theorem}{}\label{thm:funtori-derivati}
Sia \(F : \mathcal{A} \to \mathcal{B}\) un funtore additivo tra categorie
abeliane, con \(\mathcal{A}\) con abbastanza iniettivi.

Allora ogni successione esatta (corta) \(\Sigma : 0 \to X \overset{f}{\to} Y \overset{g}{\to}
Z \to 0\) in \(\mathcal{A}\) induce una successione esatta (lunga) in \(\mathcal{B}\) 
\[
  0 \to R^{0}F{(X)} \overset{R^{0}F{(f)}}{\to} R^{0}F{(Y)}
  \overset{R^{0}F{(g)}}{\to} R^{0}F{(Z)} \overset{w^{0}}{\to} R^{1}F{(X)}
  \overset{R^{1}F{(f)}}{\to} \dots
\]
\end{theorem}

\begin{proof}{}

    J^\bullet\), \(g^\bullet : J^\bullet \to L^\bullet\) in \(C{(\mathcal{A})}\)
    tali che \(0 \to I^\bullet \overset{f^\bullet}{\to} J^\bullet
    \overset{g^\bullet}{\to} L^\bullet \to 0\) è esatta e \(f^\bullet\) estende
    \(f\) e \(g^\bullet\) estende \(g\). In tal modo si ottiene
    \begin{equation}\label{eq:help_esatta_funtorib} 0 \to
    F{(I^\bullet)} \overset{F{(f^\bullet)}}{\to} F{(J^\bullet)}
    \overset{F{(g^\bullet)}}{\to} L^\bullet \to 0\end{equation} che è esatta se
    e solo se ogni \(0 \to F{(I^{i})} \to F{(J^{i})} \to F{(L^{i})} \to 0\) è
    esatta. Quest'ultima cosa è vera perché \(I^{i}, J^{i}, L^{i}\) sono iniettivi, dunque
    la successione \(0\to I^{i} \to J^{i} \to L^{i} \to 0 \) si spezza e \(J^{i}

    questo prodotto finito e quindi anche la~\eqref{eq:help_esatta_funtorib} si

\begin{corollary}{}
\begin{enumerate}[label = \arabic*.]
    \item \(R^{i}F\) è esatto al centro \(\forall i \in \mathbb{N}\) e \(R^{0}F\)
        è esatto a sinistra
    \item \(R^{0}F \cong F \iff F\) è esatto a sinistra 
    \item \(F\) è esatto se e solo se \(R^{0}F \cong F\) e \(R^{i > 0}F \cong 0\),
        ossia \(R^{0}F \cong F\) e \(R^{1}F \cong 0\) 
\end{enumerate}
\end{corollary}

\begin{proof}{}
\begin{enumerate}[label = \arabic*.]
    \item Chiaro
    \item Un'implicazione (\(\Rightarrow\)) è una diretta conseguenza di 1. Per
        \(\impliedby \) invece, sia \(X \in \mathcal{A}\) e sia \(0 \to X \to
        I^{0} \overset{d^{0}}{\to} I^{1} \to  \dots\) una risoluzione iniettiva
        di \(X\). Allora
        \begin{align*}
            R^{0}F{(X)} &= H^{0}{(F{(I^{\bullet})})} = H^{0}{(0 \to F{(I^{0})}
            \overset{F{(d^{0})}}{\to} F{(I^{1})} \to \dots)} \cong \\ &\cong
          \mathrm{Ker}{(F{(d^{0})})} \overset{\text{\(F\) esatto a sx}}{\cong}
          F{(\mathrm{Ker}{(d^{0})})} \cong F{(X)}
        \end{align*}
    \item Se \(R^{0}F \cong F\) e \(R^{i}F \cong 0\) per ogni \(i > 0\), allora
        \(R^{0}F \cong F\) e \(R^{1} F \cong 0\). Adesso, per il punto 2 \(F\) è
        esatto a sinsitra e anche a destra (poiché \(R^{1}F \cong 0\)). Resta da
        dimostrare che se \(F\) è esatto, allora \(R^{i}F \cong 0\) per ogni \(i
        > 0\). Ma effettivamente abbiamo che
        \[
          R^{i}F{(X)} = H^{i}{(F{(I^\bullet)})} = 0 \quad \forall i > 0
        \]
        poiché \(F\) è esatto e \(0 \to X \to I^{0} \to I^{1} \to \dots\) è
        esatta, per cui anche la successione \(0 \to F{(X)} \to F{(I^{0})} \to F{(I^{1})} \to
        \dots\) è esatta.
\end{enumerate}
\end{proof}
\begin{remark}{}
    I funtori derivati destri sono interessanti solo per funtori esatti a
    sinistra (ma non a destra)
\end{remark}


\begin{definition}{Funtore derivato sinistro}
    Sia \(F : \mathcal{A} \to \mathcal{B}\) un funtore additivo, con
    \(\mathcal{A}\) con abbastanza proiettivi. Allora i funtori derivati
    sinistri di \(F\) sono 
    \(
        L_{i}F := {(R^{i}F^{op})}^{op}
    \).

    In altre parole, \(\forall X \in \mathcal{A}\), presa \(p_\bullet \to X \to
    0\) una risoluzione proiettiva di \(X\) (con \(p_\bullet = \dots \to p_{1}
    \to p_{0} \to 0\) e 
    \[
      L_{i}F{(X)} := H_{i}{(F{(p_\bullet)})} = H_{i}{(\dots \to F{(p_{1})} \to
      F{(p_{0})} \to 0)}
    \]
\end{definition}

Il teorema~\ref{thm:funtori-derivati} diventa: ogni \(\Sigma : 0 \to X
\overset{f}{\to} Y \overset{g}{\to} Z \to 0\) in \(\mathcal{A}\) induce una
successione esatta in \(\mathcal{B}\), data da
\[
  \dots \to L_{1}F{(Z)} \overset{w_{1}}{\to} L_{0}F{(X)}
  \overset{L_{0}F{(f)}}{\to} L_{0}F{(Y)} \overset{L_{0}F{(g)}}{\to} L_{0}F{(Z)}
  \to 0
\]
e il corollario diventa che \(L_0 F\) è esatto a destra, e \(L_0 F \cong
F\) se e solo se \(F\) è esatto a destra.

\begin{remark}{}
    Se abbiamo \(F : \mathcal{A}^{op} \to \mathcal{B}\) additivo, con
    \(\mathcal{A}\) con abbastanza proiettivi, allora gli \(R^{i}F\) si
    calcolano con le risoluzione proiettive in \(\mathcal{A}\), ossia \(\forall
    X \in \mathcal{A}\), \(R^{i}F{(X)} = H^{i}{(F{(p_\bullet)})}\) con
    \(p_\bullet \to X \to 0\) risoluzione proiettiva.

    Presa dunque \(0 \to X \to Y \to Z \to 0\) esatta in \(\mathcal{A}\), questa
    induce \(0 \to R^{0}F{(Z)} \to R^{0}F{(Y)} \to R^{0}F{(X)} \to R^{1}F{(Z)}
    \to \dots\) esatta in \(\mathcal{B}\).
\end{remark}

	\arrow["{(R^{i}\alpha)}_Z"', from=1-4, to=2-4]
	\arrow[from=1-5, to=1-6]
	\arrow["{(R^{i+1}\alpha)}_X"', from=1-5, to=2-5]
	\arrow[from=2-1, to=2-2]

\end{remark}

\subsection{Ext / Tor}

   Sia \(\mathcal{A}\) una categoria abeliana, \(X \in \mathcal{A}\). Allora i
   funtori \(\mathcal{A}{(X, -)} : \mathcal{A} \to \mathtt{Ab}\),
   \(\mathcal{A}{(-, X)} : \mathcal{A}^{op} \to \mathtt{Ab}\) sono esatti a
   sinistra. Se \(\mathcal{A}\) ha abbastanza iniettivi si indica con 
   \[
     \mathtt{Ext}_{\mathcal{A}}^{i}{(X, -)} := R^{i}{(\mathcal{A})}{(X, -)}
   \]
   e per ogni successione esatta in \(\mathcal{A}\), \(0 \to Y^{\prime}
   \overset{f}{\to} Y \overset{g}{\to} Y'' \to 0\) ottengo una successione
   esatta in \(\mathtt{Ab}\), 
   \[
     \dots \to \mathrm{Ext}_{\mathcal{A}} ^{i}{(X, Y')} \overset{f_*}{\to}
     \mathrm{Ext}_\mathcal{A}^{i}{(X, Y)} \overset{g_*}{\to}
     \mathrm{Ext}_\mathcal{A}^{i}{(X, Y'')} \overset{w^{i}}{\to}
     \mathrm{Ext}_\mathcal{A}^{i+1}{(X, Y')} \to \dots
   \]
   con \(\mathrm{Ext}_\mathcal{A}^{i}{(X, -)} \cong \mathcal{A}{(X, -)} =
   \mathrm{Hom}_\mathcal{A}{(X, -)}\).

   Se \(\mathcal{A}\) ha abbastanza proiettivi indichiamo con
\(\overline{\mathrm{Ext}}_\mathcal{A}^{i}{(-, X)} = R^{i}{(\mathcal{A})}{(-,
   X)}\). Esplicitamente se \(0 \to Y' \overset{f}{\to} Y \overset{g}{\to} Y''
   \to 0\) è una successione esatta in \(\mathcal{A}\), allora essa induce la
   successione esatta
   \[
    \dots \to \overline{\mathrm{Ext}}_\mathcal{A}^{i}{(Y'', X)} \overset{g^{*}}{\to}
\overline{\mathrm{Ext}}_\mathcal{A}^{i}{(Y, X)} \overset{f^{*}}{\to}
\overline{\mathrm{Ext}}_\mathcal{A}^{i}{(Y'', X)}
\overset{\overline{w}^{i}}{\to}
 \overline{\mathrm{Ext}}_\mathcal{A}^{i+1}{(Y'', X)} \to \dots
   \]
   Se \(\mathcal{A} =A\mathtt{-Mod}\) scrivo \(\mathrm{Ext}_A^{i}\) e
\(\overline{\mathrm{Ext}}_A^{i}\) invece del più pedante
   \(\mathrm{Ext}_{A\mathtt{-Mod}} ^{i}\) e
\(\overline{\mathrm{Ext}}_{A\mathtt{-Mod}} ^{i}\).

Inoltre, \(\forall M \in \mathtt{Mod-}A\), \(M \otimes_A - : A\mathtt{-Mod} \to
\mathtt{Ab}\) è esatto a destra. Allora si indica \(\mathrm{Tor}_i^{A}{(M, -)}
:= L_{i}{(M \otimes_A -)}\) e per ogni \(0 \to N' \overset{f}{\to} N
\overset{g}{\to} N'' \to 0\) esatta in \(A\mathtt{-Mod}\), essa induce
\[
  \dots \to \mathrm{Tor}_i^{A}{(M, N')} \overset{f_*}{\to}
  \mathrm{Tor}_i^{A}{(M, N)} \overset{g_*}{\to} \mathrm{Tor}_i^{A} \to
  \mathrm{Tor}_{i+1} ^{A}{(M, N')} \to \dots
\]
e il funtore \(\mathrm{Tor}_0{(M, -)} \cong M \otimes_A -\).

Dualmente, se \(N \in A\mathtt{-Mod}\), allora \(- \otimes_A N : \mathtt{Mod-}A
\to \mathtt{Ab}\) è esatto a destra, dunque \(\overline{\mathrm{Tor}}_i^{A}{(-,
N)} := L_{i}{(- \otimes_A N)}\).


Sia ora \(\mathcal{A}\) una categoria abeliana con abbastanza iniettivi, \(f : X
\to X'\) un morfismo di \(\mathcal{A}\). Allora \(\forall Y \in \mathcal{A}\),
\(f^{*} : \mathcal{A}{(X', Y)} \to \mathcal{A}{(X, Y)}\) definisce una
trasformazione naturale \(f^{*} : \mathcal{A}{(X', -)} \to \mathcal{A}{(X, -)}\)
che induce una trasformazione naturale \(R^{i}f^{*} =: f^{*} :
\mathrm{Ext}_\mathcal{A}^{i}{(X', -)} \to \mathrm{Ext}^{i}_\mathcal{A}{(X, -)}\)
per ogni \(i \in \mathbb{N}\). Si ottiene dunque un funtore
\(\mathrm{Ext}_\mathcal{A}^{i}{(-, =)} : \mathcal{A}^{op} \times \mathcal{A} \to
\mathtt{Ab}\) tale che \(\mathrm{Ext}_\mathcal{A}^{0}{(-, =)} \cong
\mathcal{A}{(-, =)}\). 

Analogamente, se \(\mathcal{A}\) ha abbastanza proiettivi, ottengo, per ogni \(i
\in \mathbb{N}\), un funtore \(
  \overline{\mathrm{Ext}}_\mathcal{A}^{i} : \mathcal{A}^{op} \times \mathcal{A}
  \to \mathtt{Ab} \text{ tale che } \overline{\mathrm{Ext}}_\mathcal{A}^{0}{(-,
  =)} \cong \mathcal{A}{(-, =)}
\). 

Analogamente (sorry se qui noi si continua a fare analogie), per ogni anello
\(A\) si ottengono funtori \(\mathrm{Tor}_i^{A}{(-, =)}\),
\(\overline{\mathrm{Tor}}_i^{A}{(-, =)} : \mathtt{Mod-}A \times A\mathtt{-Mod}
\to \mathtt{Ab}\) tali che \(\mathrm{Tor}_0^{A}{(-, =)} \cong
\overline{\mathrm{Tor}}_0^{A}{(-, =)} \cong - \otimes_A =\).

\begin{proposition}{}
    Siano \(F, F', F'' : \mathcal{A} \to \mathcal{B}\) funtori additivi,
    \(\mathcal{A}\) con abbastanza iniettivi, \(F' \overset{\alpha}{\to} F
    \overset{\beta}{\to} F''\) trasformazioni naturali tali che \(\forall I \in
    \mathcal{A}\) iniettivo, la successione \(0 \to F'{(I)}
    \overset{\alpha_I}{\to} F{(I)} \overset{\beta_I}{\to} F''{(I)} \to 0\) è
    esatta in \(\mathcal{B}\)\dots
    
    Allora, per ogni \(X \in \mathcal{A}\), la seguente successione è esatta in
    \(\mathcal{B}\):
    \[
      \dots \to R^{i}F'{(X)} \overset{{(R^{i}\alpha)}_X}{\to} R^{i}F{(X)}
      \overset{{(R^{i}\beta)}_X}{\to} R{(i)}F''{(X)} \to R^{i+1}F''{(X)} \to \dots
    \]
\end{proposition}

\begin{proof}{}
    \(0 \to X \to I^\bullet\) è una risoluzione iniettiva di \(X\). Allora
    \(R^{i}F^{(', '')}{(X)} := H^{i}{(F^{{(', '')}}{(I^\bullet)})}\). Allora la
    successione 
    \[
      0 \to F'{(I^\bullet)} \overset{\alpha_{I^\bullet} }{\to} F{(I^\bullet)}
      \overset{\beta_{I^\bullet} }{\to} F''{(I^\bullet)} \to 0
    \]
    è esatta in \(C{(\mathcal{B})}\) perché per ipotesi \(0 \to F'{(I^{n})}
    \overset{\alpha_{I^{n}} }{\to} F{(I^{n})} \overset{\beta_{I^{n}} }{\to}
    F''{(I^{n})}\) lo è in \(\mathcal{B}\) e appplico la successione esatta di
    coomologia.
\end{proof}

    \]
    e analogamente, partendo dalla successione esatta corta \(0 \to Y' \to Y \to
    Y'' \to 0\) esatta in \(\mathcal{A}\), allora \(\forall X \in \mathcal{A}\)
    la seguente successione è esatta in \(\mathtt{Ab}\):
\end{corollary}


Sia \(\mathcal{A}\) una categoria abeliana con abbastanza iniettivi. Allora
\(\mathrm{Ext}_\mathcal{A}^{i}{(X, -)} := R^{i}\mathcal{A}{(X, -)}\).
Analogamente se \(\mathcal{A}\) ha abbastanza proiettivi
\(\overline{\mathrm{Ext}}_{\mathcal{A}} ^{i}{(-, X)} := R^{i}\mathcal{A}{(-,
X)}\).

\begin{theorem}{}
    Se \(\mathcal{A}\) ha abbastanza iniettivi e proiettivi, allora
    \(\mathrm{Ext}_\mathcal{A}^{i} {(-, =)} \cong
    \overline{\mathrm{Ext}}_\mathcal{A}^{i} {(-, =)}\), \(\forall i \in
    \mathbb{N}\).
\end{theorem}
\begin{remark}{}
    Sapevamo già che entrambi i (bi)funtori mandano successioni esatte corte in
    ogni argomento in successioni esatte lunghe.
\end{remark}
\begin{proof}{}
    È necessario trovare un isomorfismo naturale \(\theta^{i} :
    \mathrm{Ext}^{i}{(-, =)} \to \overline{\mathrm{Ext}}^{i}{(-, =)}\) per ogni \(i \in
    \mathbb{N}\). Dimostriamone l'esistenza per induzione su \(i\). Se \(i=0\)
    siamo a cavallo, poiché \(\mathrm{Ext}^{0}{(-, =)} \cong \mathcal{A}{(-, =)}
    \cong \overline{\mathrm{Ext}}^{0}{(-, =)}\). Supponiamo ora che esista un
    \(\theta^{i}\) e dimostriamo che esiste \(\theta^{i+1}\).

    A tale scopo sia \(X, Y \in \mathcal{A}\). Sia allora \(0 \to Y \to I \to Z
    \to 0\) una successione esatta con \(I\) iniettivo (\(I\) esiste perché
    \(\mathcal{A}\) ha abbastanza iniettivi, \(Z\) è il coker). Applico le
    successioni esatte lunghe

\[\begin{tikzcd}
    && \mathrm{CoKer}{(g_*)} & R^{i+1} \mathcal{A}{(X, -)}{(I)} \\
	\mathrm{Ext}^{i}{(X,I)} & \mathrm{Ext}^{i}{(X,Z)} &
    \mathrm{Ext}^{i+1}{(X,Y)} & \mathrm{Ext}^{i+1}{(X,I)} = 0 \\
	\overline{\mathrm{Ext}}^{i}{(X,I)} & \overline{\mathrm{Ext}}^{i}{(X,Z)} &
    \overline{\mathrm{Ext}}^{i+1}{(X,Y)} & \overline{\mathrm{Ext}}^{i+1}{(X,I)}
    = 0
	\arrow["{g_*}", from=2-1, to=2-2]
	\arrow["{\theta^i_{X,I}}"', from=2-1, to=3-1]
	\arrow[from=2-2, to=2-3]
	\arrow["{\theta^i_{X,Z}}"', from=2-2, to=3-2]
	\arrow["\cong"{marking, allow upside down}, draw=none, from=2-3, to=1-3]
	\arrow[from=2-3, to=2-4]
	\arrow["{\theta^{i+1}_{X,Y}}"', dashed, from=2-3, to=3-3]
	\arrow["{=}"{marking, allow upside down}, draw=none, from=2-4, to=1-4]
	\arrow["{g_*}", from=3-1, to=3-2]
	\arrow[from=3-2, to=3-3]
	\arrow[from=3-3, to=3-4]
\end{tikzcd}\]
    Per la proprietà universale del \(\mathrm{CoKer}\), \(\exists !
    \theta^{i+1}_{X,Y} \) che rende il diagramma commutativo e si verifica che
    \(\theta^{i+1}\) è naturale. Allora se \(\theta^{i}\) è iso, anche
    \(\theta^{i+1}\) lo è.
\end{proof}

\begin{example}{}
    Sia \(A\) un PID non campo. Allora \(\forall M \in A\mathtt{-Mod}\), esiste
    una risoluzione proiettiva \(0 \to P_{1} \to P_{0} \to M \to 0\) (perché un
    sottomodulo di un modulo libero è libero e proiettivo equivale libero).
    Analogamente esiste la risoluzione iniettiva \(0 \to M \to I^{0} \to I^{1}
    \to 0\) (poiché iniettivo significa divisibile e un quozione di un
    divisibile è divisibile). Quindi \(\forall F : A\mathtt{-Mod} \to
    \mathcal{A}\) additivo con \(\mathcal{A}\) una categoria abeliana, \(R^{i}F
    \cong 0 \cong L_{i} F\) per ogni \(i > 1\). 

    Allora come posso trovare \(\mathrm{Ext}_A^{i}\) e \(\mathrm{Tor}_i^{A}\),
    per \(i = 0, 1\), sui moduli ciclici (e quindi sui moduli f.g.)?
    \[
      \mathrm{Ext}_A^{i}{(A, -)} = R^{i}\mathrm{Hom}_A{(A,-)} \cong R^{i}
      \mathrm{id}_{A\mathtt{-Mod}} \cong \begin{cases}{}
          \mathrm{id}_{A\mathtt{-Mod}} & \text{ se }i = 0\\
          0 & \text{ se } i > 0
      \end{cases}
    \]
    e a analogamente
    \[
      \mathrm{Tor}_i^{A}{(A, -)} \cong \begin{cases}{}
          \mathrm{id}_{A\mathtt{-Mod}} & \text{ se } i = 0 \\
          0 & \text{ se } i > 0
      \end{cases}
    \]

    Sia ora \(0 \neq a \in A \sminus A^{*}\) nel PID. Allora per ogni \(M \in
    A\mathtt{-Mod}\) ci chiediamo cosa siano \(\mathrm{Ext}_a^{i}{(A / {(a)},
    M)}\) e \(\mathrm{Tor}{(A / {(a)}, M)}\).
    Consideriamo la risoluzione proiettiva di \(A / {(a)}\), \( 0 \to A
    \overset{-\cdot a}{\to} A \to A / {(a)} \to 0\). Allora
    \begin{align*}
        \mathrm{Ext}_\mathcal{A}^{i} {(A / {(a)}, M )} &\cong
      H^{i}{\left(\mathrm{Hom}_A{\left( {(0 \to A \overset{-\cdot a}{\to} A \to 0)},
M\right)} \right)} \cong \\ &\cong H^{i}{\left( 0 \to M \cong \mathrm{Hom}_A{(A, M)}
      \overset{\cdot a}{\to} M \to 0 \right)} \cong \begin{cases}{}
          \mathrm{Ker}{(M \overset{-\cdot a}{\to} M)} & \text{ se } i = 0 \\
          \mathrm{CoKer}{(M \overset{-\cdot a}{\to} M)} & \text{ se } i = 1
      \end{cases}
    \end{align*}
    E abbiamo un risultato analogo per il \(\mathrm{Tor}_i^{A}{(A / {(a)}, M)}\).

    Se \(M = A\), allora \(\mathrm{Ker}{(A \overset{-\cdot a}{\to} A)} = 0\) e
    \(\mathrm{CoKer}{(A \overset{-\cdot a}{\to} A)} = A / {(a)}\). Quindi
    \(\mathrm{Hom}_A{(A/{(a)}, A)} = 0 = \mathrm{Tor}_1^{A}{(A / {(a)}, A)}\) e
        \begin{align*}
            \mathrm{Ext}^{1}_A{(A / {(a)}, A)} &\cong A /{(a)} \\
            &\cong A / {(a)} \otimes_A A
        \end{align*}

    Sia \(M = A / {(b)}\), con \(0 \neq b \in A \sminus A^{*}\).
    Chi sono allora \(\mathrm{Ker}\) e \(\mathrm{Coker}\) di \(A / {(b)}
    \overset{-*a}{\to} A / {(b)}\)? Prendiamo \(d := \mathrm{mcd}{(a, b)}\).
    Allora \(a = da'\) e \(b = db'\) e \(mcd{(a', b')} = 1\). A questo punto è
    facile vedere che \(\mathrm{Im}{\left( A / {(a)} \overset{\cdot a}{\to} A /
    {(b)}\right)} = {(a,b)}/{(b)} = {(d)}/{(b)} = {(d)}/{(db')}\), dunque il
    \(\mathrm{Coker}\) di \(A/{(b)} \overset{-\cdot a}{\to} A/{(b)}\) è 
    \[
      {\left( \frac{A}{{(b)}} \right)} / {\left( \frac{{(d)}}{{(b)}} \right)} \cong A /
      {(d)}
    \]
    Per il Ker, notiamo che
    \[
        \mathrm{Im} {(- \cdot a)} \cong {\left( \frac{A}{{(b)}} \right)} /
        {\left( \frac{{(b'')}}{{(b)}} \right)} = A / {(b'')} \implies b'' = b'
    \]
    Da cui \(\mathrm{Ker}{(- \cdot a)} = \frac{{(b')}}{{(b)}} =
    \frac{{(b')}}{{(db')}} = A / {(d)}\), per cui
    \[
      \mathrm{Ext}_A^{i}{\left( A / {(a)}, A / {(b)} \right)} \cong
      \mathrm{Tor}_i^{A}{\left( A / {(a)}, A / {(b)} \right)} \cong A / {(d)} =
      A / {(a, b)} \quad \forall i \in \{0, 1\} 
    \]
    (per cui se \(a\) e \(b\) sono coprimi viene 0)
\end{example}

\begin{example}{}
    Sia \(A\) un PID non campo, \({(p)} = J \subseteq A \) un ideale massimale,
    con \({(p)} \in A\) irriducibile. Sia \(\overline{A} := A / J^2\). Allora
    \(\overline{J} := J / J^2 \subseteq \overline{A} \) è ideale, e in

    \(\overline{A} \overset{-*\overline{p}}{\to} \overline{A}\) ha immagine e
    nucleo \(\overline{J}\), e conucleo \(\overline{A} / \overline{J} =
    \overline{J}\). 

    Quindi \(\dots \to \overline{A} \overset{-\cdot \overline{p}}{\to}
    \overline{A} \overset{-\cdot \overline{p}}{\to} \overline{A} \to
    \overline{J} \to 0\) è risoluzione proiettiva di \(\overline{J}\). Allora

        \mathrm{Ext}_A^{i}{(\overline{J}, \overline{J})} &= H^{i}{\left(
      \mathrm{Hom}_{\overline{A}} {\left( {\left( \dots \to \overline{A}

    Analogamente \(\mathrm{Tor}_i^{\overline{A}}{(\overline{J}, \overline{J})}

\begin{definition}{Dimensione iniettiva/proiettiva}
Sia \(\mathcal{A}\) una categoria abeliana con abbastanza iniettivi. Sia \(0\neq X \in
\mathcal{A}\). Allora la \emph{dimensione iniettiva} di \(X\) è
\[
  \mathrm{idim}{(X)} := \min \{n \in \mathbb{N} : \exists \text{ risol. inj. di
  \(X\) di lunghezza \(n\)}\} 
\]
e la \emph{dimensione proiettiva} di \(X\), \(\mathrm{pdim}{(X)}\)  è la stessa cosa per risoluzioni
proiettive.

È facile vedere che \(\mathrm{idim}{(X)} = \sup \{ n \in \mathbb{N} :
\mathrm{Ext}_\mathcal{A}^{n}{(-, X)} \neq 0\} \). Motivati dalla forma di questa
espressione equivalente, si definisce la \emph{dimensione globale} di
\(\mathcal{A}\) come
\[
  \mathrm{gldim}{(\mathcal{A})} := \sup \{n \in \mathbb{N} :
  \mathrm{Ext}_{\mathcal{A}}^{n}{(-, =)} \not\cong 0 \} 
\]
e se \(\mathcal{A}\) ha abbastanza iniettivi, \(\mathrm{gldim}{(\mathcal{A})} =
\sup \{ \mathrm{idim}{(X)} : X \in \mathcal{A}\} \). In particolare dato \(A\)
anello, si definiscono
\[
  \mathrm{lgldim}{(A)} := \mathrm{gldim}{(A\mathtt{-Mod})} \text{  e  }
  \mathrm{rgldim}{(A)} := \mathrm{gldim}{(\mathtt{Mod-}A)}
\]
\end{definition}

\begin{remark}{}
chiaramente se \(A\) è un anello comutativo o noetheriano (sia a dx che a sx)
\(\mathrm{lgldim}{(A)} = \mathrm{rgldim}{(A)} =: \mathrm{gldim}{(A)}\). 

Inoltre se \(A\) è commutativo, neotheriano e locale (i.e. con un solo ideale
massimale \(J\)), Allora \(\dim\footnote{La dimensione (di KRULL) di \(A\neq 0 \) commutativo è definita come
    \(\dim A = \sup \{n \in \mathbb{N} : \exists P_{0} \subsetneq p_{1} \subsetneq \dots
\subsetneq p_n \text{ ideali primi di \(A\)}\}\)} A
\le \dim_{A/J}{(J / J^2)} < \infty\) e \(A\)
si dice \emph{regolare} se l'uguaglianza si realizza, e dunque non è \(\le \) ma
\(=\). Si dimostra che \(A\) è regolare se e solo se \(\mathrm{gldim}{(A)} <
\infty\) e in tal caso \(\mathrm{gldim}{(A)} = \dim A\) 
\end{remark}

Una definizione alternativa di \(\mathrm{Ext}_\mathcal{A}^{i}\) (anche quando
\(\mathcal{A}\) non ha abbastana iniettivi o proiettivi) è la seguente. Sia
\(\mathcal{A}\) una categoria abeliana. Allora \(\forall X, Y \in \mathcal{A}\)
si definisce
\[
  E^{1}_\mathcal{A}{(X, Y)} := \{ \varepsilon : 0 \to Y \overset{f}{\to} Z
  \overset{g}{\to} X \to 0 \text{ successione esatta}\}/_\sim 
\]
dove \(\varepsilon \sim \varepsilon' : 0 \to Y \overset{f'}{\to} Z'
\overset{g'}{\to} X \to 0\) se e solo se \(\exists h : Z \to Z'\) tale che \(g
\circ f =f'\) e \(g' \circ h = g\), o in altre parole il seguente diagramma
commuta
\[ \begin{tikzcd}
	0 & Y & Z & X & 0 \\
	0 & Y & {Z'} & X & 0
	\arrow[from=1-1, to=1-2]
	\arrow["f", from=1-2, to=1-3]
	\arrow["{1_Y}"', from=1-2, to=2-2]
	\arrow["g", from=1-3, to=1-4]
	\arrow["h"', dashed, from=1-3, to=2-3]
	\arrow[from=1-4, to=1-5]
	\arrow["{1_Y}"', from=1-4, to=2-4]
	\arrow[from=2-1, to=2-2]
	\arrow["{f'}", from=2-2, to=2-3]
	\arrow["{g'}", from=2-3, to=2-4]
	\arrow[from=2-4, to=2-5]
\end{tikzcd} \]
\begin{remark}{}
    \(h\) è isomorfismo (per il lemma del seprente), dunque \(\sim \) è una
    relazione di equivalenza.
\end{remark}

Se \(\mathcal{A}\) ha abbastanza iniettivi o proiettivi allora si dimostra che
la funzione
\begin{align*}
    \mathrm{E}_\mathcal{A}^{1}{(X, Y)} &\to \mathrm{Ext}_\mathcal{A}^{1}{(X, Y)}
    \\
    [\varepsilon] \longmapsto \omega^{0}{(\varepsilon)}{(1_X)}
\end{align*}
applicando \(\mathcal{A}{(X, -)}\) a \(\varepsilon\) trovo successioni esatte
corte in \(\mathtt{Ab}\) 
\[
  \mathcal{A}{(X, Z)} \overset{g_*}{\to} \mathcal{A}{(X, X)}
  \overset{w^{0}{(\varepsilon)}}{\to} \mathrm{Ext}^{1}_\mathcal{A}{(X, Y)} \to
  \dots
\]
Si dimostra che \(\mathrm{E}^{1}_\mathcal{A}{(X, Y)}\) ha struttura di gruppo
abeliano (naturale) in ogni caso dove 0 è la classe di \(\sim \) delle successioni esatte
corte che si spezzano.
Più in generale si definisce, \(\forall n \ge  1\),
\[
  \mathrm{E}_\mathcal{A}^{n}{(X, Y)} := \{0 \to Y \to Z_{1} \to \dots \to Z_n
  \to X \to 0 \text{ esatta}\}/_\sim 
\]
cno \(\sim \) relazione d'equivalenza generata da \(\sim'\) definita come
\(\varepsilon \sim' \varepsilon' : 0 \to Y \to Z_{1}' \to \dots Z'_n \to X \to
0\) e se e solo se \(\exists h_i : Z_{i} \to Z'_i\) per ogni \(i = 1..n\)  che
rendono il diagramma commutativo.


    Dato \(F : \mathcal{A} \to \mathcal{B}\) additivo tra categorie abeliane, si
    possono definire \(R^{i}F : \mathcal{A} \to \mathcal{B}\) tali che vale il 
    teorema fondamentale dei funtori derivati. Se \(\exists  \mathcal{J}
    \subseteq \mathcal{A} \) sottocategoria piena tale che 
\begin{enumerate}[label = \arabic*)]
    \item \(\forall X \in \mathcal{A}\), \(\exists 0 \to X \to I^{0} \to I^{1}
        \to \dots\) risoluzione con \(I^{i} \in \mathcal{J}\) 
    \item Se la successione \(0 \to I^{0} \to I^{1} \to \dots\) è esatta in \(\mathcal{A}\) con
        \(I^{i} \in \mathcal{J}\), allora \(0 \to F{(I^{0})} \to F^{I^{1}} \to
        \dots\) è esatta
\end{enumerate}
poi si definisce \(R^{i}F{(X)} := H^{i}{(F{(I)})}\) dove \(0 \to X \to I\) è
esatta e \(I^{i} \in \mathcal{J}\). Se \(\mathcal{A}\) ha abbastanza inietivi si
può sempre prendere la sottocatoria data da
\[
  \mathrm{Ob}{(J)} = \{X \in \mathcal{A} : X \text{ iniettivo}\} 
\]

\begin{example}{}
    Se \(A\) è un anello, allora si può calcolare \(\mathrm{Tor}_i^{A}\) usando
    le risoluzioni \emph{piatte}.
\end{example}
